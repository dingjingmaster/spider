cmake_minimum_required(VERSION 3.16)
project(spider VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SPIDER_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3>=3.35)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c>=0.15)
find_package(Threads REQUIRED)
find_package(OpenSSL 1.1.1 REQUIRED)
if(SPIDER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
enable_testing()
add_subdirectory(src)
