set(LEXBOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_STATIC ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/lexbor EXCLUDE_FROM_ALL)
add_library(civetweb STATIC third_party/civetweb/src/civetweb.c)
target_include_directories(civetweb PUBLIC third_party/civetweb/include)
target_compile_definitions(civetweb PRIVATE NO_SSL NO_CGI NO_CACHING USE_WEBSOCKET)
target_link_libraries(civetweb PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(spider_core STATIC
  common/common.c rule/rule.c extract/extract.c storage/storage.c
  scheduler/scheduler.c http/http.c artifact/artifact.c engine/engine.c
  browser/browser.c service/service.c)
target_include_directories(spider_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} third_party/lexbor/source)
target_compile_definitions(spider_core PUBLIC _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
target_compile_options(spider_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
target_link_libraries(spider_core PUBLIC PkgConfig::CURL PkgConfig::SQLITE PkgConfig::JSON Threads::Threads OpenSSL::Crypto lexbor_static m)
add_executable(spider main/main.c web/web.c)
target_link_libraries(spider PRIVATE spider_core civetweb)
target_compile_options(spider PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
set_target_properties(spider PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_custom_target(spider_ui ALL COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/web/ui ${CMAKE_BINARY_DIR}/ui)
add_dependencies(spider spider_ui)
install(TARGETS spider RUNTIME DESTINATION bin)
install(DIRECTORY web/ui/ DESTINATION share/spider/ui)
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE spider_core)
target_compile_options(test_core PRIVATE -UNDEBUG)
add_test(NAME core COMMAND test_core)
add_executable(test_storage tests/test_storage.c)
target_link_libraries(test_storage PRIVATE spider_core)
target_compile_options(test_storage PRIVATE -UNDEBUG)
add_test(NAME storage COMMAND test_storage)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:spider>)
  set_tests_properties(integration PROPERTIES TIMEOUT 90)
endif()
